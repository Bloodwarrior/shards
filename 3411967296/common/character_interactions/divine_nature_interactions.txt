overgrowth_interaction = {
    category = interaction_category_religion
    icon = magic_positive
    interface_priority = 40

    desc = overgrowth_interaction_desc

    target_type = title
    target_filter = recipient_domain_titles

    is_shown = {
        scope:actor = {
            is_a_divinity = yes
            has_character_modifier = nature_domain_modifier
        }
        scope:recipient = { is_landed = yes }
    }

    can_be_picked = {
        scope:target = { tier = tier_county }
    }

    on_accept = {
        scope:target = {
            add_county_modifier = {
                modifier = overgrowth_modifier
                years = 10
            }
        }
    }

    auto_accept = yes
    cost = { piety = 100 }
}

natures_vigor_interaction = {
    category = interaction_category_religion
    icon = magic_positive
    interface_priority = 80

    desc = natures_vigor_interaction_desc

    is_shown = {
        scope:actor = {
            is_a_divinity = yes
            has_character_modifier = nature_domain_modifier
        }
    }

    on_accept = {
        scope:recipient = {
            add_character_modifier = {
                modifier = natures_vigor_modifier
                years = 20
            }
            add_opinion = {
                target = scope:actor
                modifier = blessed_me_opinion
            }
        }
    }

    auto_accept = yes
    cost = { piety = 100 }
}
