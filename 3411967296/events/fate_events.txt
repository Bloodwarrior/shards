# Unravel Destiny Event
fate_events.0006 = {
	title = fate_events.0006.title
	desc = fate_events.0006.desc
	theme = intrigue

	left_portrait = root
	right_portrait = scope:deity_scope

	option = {
		name = fate_events.0006.a
		add_character_modifier = {
			name = unraveled_destiny_modifier
			duration = 7300 # 20 years
		}
		add_opinion = {
			target = scope:deity_scope
			modifier = cursed_me_opinion
		}
		every_held_title = {
			root = {
				remove_claim = prev
			}
		}
		if = {
			limit = {
				is_ruler = yes
			}
			primary_title = {
				if = {
					limit = {
						tier > 1 # Don't want to make them unlanded if they are just a count
					}
					destroy_title = yes
				}
			}
		}
	}
}
